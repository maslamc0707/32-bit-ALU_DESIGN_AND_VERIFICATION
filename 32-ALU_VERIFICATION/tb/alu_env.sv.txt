
class alu_env extends uvm_env;
  `uvm_component_utils(alu_env)

  alu_agent agent;
  alu_scoreboard scoreboard;
  alu_env_config m_env_cfg;

  function new (string name, uvm_component parent);
    super.new(name, parent);
  endfunction

  virtual function void build_phase (uvm_phase phase);
    super.build_phase(phase);

`uvm_info("ENV","inside alu build phase", UVM_LOW)

    if (!uvm_config_db #(alu_env_config)::get(this, "", "env_config", m_env_cfg))
      `uvm_fatal("ENV", "failed to get env_config")

    uvm_config_db #(alu_agent_config)::set(this, "agent*", "agent_config", m_env_cfg.m_cfg);

    agent = alu_agent::type_id::create("agent", this);

   if(m_env_cfg.has_scoreboard)begin

scoreboard = alu_scoreboard::type_id::create("scoreboard", this);
end
  endfunction

  virtual function void connect_phase (uvm_phase phase);
    if (m_env_cfg.has_scoreboard)
      agent.mon.ap.connect(scoreboard.ap_imp);
  endfunction

endclass
